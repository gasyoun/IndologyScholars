Sub NNNN()

Dim n As Byte, i, S As String
Open "E:\pwg.txt" For Binary Access Read As 1
i = 1
S = ""
k = 0

While i <= LOF(1)
Get #1, i, n
If n <> 10 Then
S = S + ChrW(n)
Else
If S = "<LEND>" Then
k = k + 1
End If
S = ""
End If
i = i + 1
Wend
MsgBox LOF(1) & "/" & i
MsgBox k
Close #1
End Sub
